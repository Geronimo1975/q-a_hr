/* JavaScript pentru a elimina orice popup text rămas și a îmbunătăți comportamentul butonului de chat */
// Adaugă acest cod în fișierul tău JavaScript
document.addEventListener('DOMContentLoaded', function() {
    // Elimină toate popup-urile existente
    const popups = document.querySelectorAll('.chat-popup, .popup-text, .chat-bubble');
    popups.forEach(popup => {
        popup.style.display = 'none';
        popup.remove(); // Elimină complet elementul
    });
    
    // Îmbunătățește butonul de chat
    const chatButton = document.querySelector('.chat-launcher');
    if (chatButton) {
        // Înlocuiește conținutul butonului cu o iconiță simplă (dacă nu există deja)
        if (!chatButton.querySelector('.chat-icon')) {
            chatButton.innerHTML = `
                <span class="chat-icon">
                    <svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round">
                        <path d="M21 15a2 2 0 0 1-2 2H7l-4 4V5a2 2 0 0 1 2-2h14a2 2 0 0 1 2 2z"></path>
                    </svg>
                </span>
            `;
        }
    }
});
