This prompt provides detailed instructions to transform your existing chatbot into a professional Telekom-style chatbot with topic buttons and a modern floating interface.
Redesign Objectives
Transform the current chatbot implementation into a sleek, professional chat interface with:

Telekom Brand Styling - Implement the magenta color scheme (#E20074) with clean, modern aesthetics
Topic Selection Buttons - Add clickable topic buttons for frequently asked questions
Improved User Experience - Enhanced message styling, typing indicators, and smooth animations

3. JavaScript Implementation
Update the JavaScript to handle the new UI elements:
javascriptdocument.addEventListener('DOMContentLoaded', function() {
    // DOM Elements
    const chatWidget = document.getElementById('chat-widget');
    const chatLauncher = document.getElementById('chat-launcher');
    const minimizeButton = document.getElementById('minimize-chat');
    const closeButton = document.getElementById('close-chat');
    const chatMessages = document.getElementById('chat-messages');
    const chatForm = document.getElementById('chat-form');
    const userMessageInput = document.getElementById('user-message');
    const typingIndicator = document.getElementById('typing-indicator');
    const uploadForm = document.getElementById('upload-form');
    const fileStatus = document.getElementById('file-status');
    const uploadModal = document.getElementById('upload-modal');
    const closeUploadModal = document.getElementById('close-upload-modal');
    const sessionId = '{{ session_id }}';
    
    // Initialize Chat
    function initChat() {
        // Toggle chat visibility
        chatLauncher.addEventListener('click', () => {
            chatWidget.classList.remove('minimized');
            chatLauncher.classList.add('hidden');
            scrollToBottom();
        });
        
        minimizeButton.addEventListener('click', () => {
            chatWidget.classList.add('minimized');
            chatLauncher.classList.remove('hidden');
        });
        
        closeButton.addEventListener('click', () => {
            chatWidget.classList.add('minimized');
            chatLauncher.classList.remove('hidden');
        });
        
        // Handle topic button clicks
        document.querySelectorAll('.topic-button').forEach(button => {
            button.addEventListener('click', function() {
                const topic = this.dataset.topic;
                let question = '';
                
                switch(topic) {
                    case 'Application':
                        question = 'How do I apply to become a delivery driver?';
                        break;
                    case 'Requirements':
                        question = 'What are the requirements to become a delivery driver?';
                        break;
                    case 'Payment':
                        question = 'How much can I earn as a delivery driver?';
                        break;
                    case 'Vehicle':
                        question = 'Do I need my own vehicle?';
                        break;
                    case 'Other':
                        question = 'What areas do you service?';
                        break;
                }
                
                userMessageInput.value = question;
                chatForm.dispatchEvent(new Event('submit'));
            });
        });
        
        // Document upload modal
        document.getElementById('upload-document-btn').addEventListener('click', () => {
            uploadModal.classList.add('active');
        });
        
        closeUploadModal.addEventListener('click', () => {
            uploadModal.classList.remove('active');
        });
        
        // Scroll to latest message
        scrollToBottom();
    }
    
    // Scroll chat to bottom
    function scrollToBottom() {
        chatMessages.scrollTop = chatMessages.scrollHeight;
    }
    
    // Add message to chat
    function addMessage(content, role, timestamp = null) {
        const messageEl = document.createElement('div');
        messageEl.classList.add('message');
        
        if (role === 'user') {
            messageEl.classList.add('user-message');
            messageEl.textContent = content;
        } else if (role === 'assistant') {
            messageEl.classList.add('assistant-message');
            messageEl.innerHTML = content.replace(/\n/g, '<br>');
        } else if (role === 'system') {
            messageEl.classList.add('system-message');
            messageEl.textContent = content;
        }
        
        // Add timestamp
        const timeEl = document.createElement('div');
        timeEl.classList.add('message-time');
        
        if (timestamp) {
            const date = new Date(timestamp);
            timeEl.textContent = date.toLocaleTimeString([], {hour: '2-digit', minute:'2-digit'});
        } else {
            timeEl.textContent = new Date().toLocaleTimeString([], {hour: '2-digit', minute:'2-digit'});
        }
        
        messageEl.appendChild(timeEl);
        
        // Insert before typing indicator
        chatMessages.insertBefore(messageEl, typingIndicator);
        scrollToBottom();
    }
    
    // Show typing indicator
    function showTypingIndicator() {
        typingIndicator.style.display = 'block';
        scrollToBottom();
    }
    
    // Hide typing indicator
    function hideTypingIndicator() {
        typingIndicator.style.display = 'none';
    }
    
    // Send message to API
    async function sendMessage(message) {
        addMessage(message, 'user');
        showTypingIndicator();
        
        try {
            const response = await fetch('/chat/api/send_message/', {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json',
                },
                body: JSON.stringify({
                    message: message,
                    session_id: sessionId
                })
            });
            
            const data = await response.json();
            hideTypingIndicator();
            
            if (data.success) {
                addMessage(data.message, 'assistant', data.timestamp);
            } else {
                addMessage(data.error || 'Sorry, I encountered an error processing your request.', 'system');
            }
        } catch (error) {
            hideTypingIndicator();
            addMessage('Sorry, there was a network error. Please try again.', 'system');
            console.error('Error sending message:', error);
        }
    }
    
    // Handle chat form submission
    chatForm.addEventListener('submit', function(e) {
        e.preventDefault();
        
        const message = userMessageInput.value.trim();
        if (!message) return;
        
        userMessageInput.value = '';
        sendMessage(message);
    });
    
    // Handle file upload
    uploadForm.addEventListener('submit', async function(e) {
        e.preventDefault();
        
        const fileInput = document.getElementById('document');
        if (!fileInput.files.length) {
            showFileStatus('Please select a file', false);
            return;
        }
        
        const file = fileInput.files[0];
        if (!file.name.toLowerCase().endsWith('.pdf')) {
            showFileStatus('Only PDF files are allowed', false);
            return;
        }
        
        const formData = new FormData();
        formData.append('document', file);
        
        showFileStatus('Uploading document...', true);
        
        try {
            const response = await fetch('/chat/api/upload_document/', {
                method: 'POST',
                body: formData
            });
            
            const data = await response.json();
            
            if (response.ok) {
                showFileStatus('Document uploaded successfully: ' + file.name, true);
                uploadModal.classList.remove('active');
                addMessage(`Document uploaded: ${file.name}`, 'system');
            } else {
                showFileStatus(data.error || 'Error uploading document', false);
            }
        } catch (error) {
            showFileStatus('Network error during upload', false);
            console.error('Error uploading document:', error);
        }
    });
    
    // Show file status
    function showFileStatus(message, isSuccess) {
        fileStatus.textContent = message;
        fileStatus.style.display = 'block';
        
        fileStatus.className = isSuccess ? 'status-success' : 'status-error';
        
        setTimeout(() => {
            fileStatus.style.display = 'none';
        }, 5000);
    }
    
    // Initialize the chat interface
    initChat();
});
