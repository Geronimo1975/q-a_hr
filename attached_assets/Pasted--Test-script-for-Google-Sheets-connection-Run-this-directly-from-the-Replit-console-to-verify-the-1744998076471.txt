"""
Test script for Google Sheets connection
Run this directly from the Replit console to verify the connection
"""

import os
import sys
import json
from pathlib import Path

# Add the project directory to the path so we can import from utils
BASE_DIR = Path(__file__).resolve().parent
sys.path.append(str(BASE_DIR))

try:
    # Try to import Django settings
    os.environ.setdefault('DJANGO_SETTINGS_MODULE', 'driver_recruitment.settings')
    import django
    django.setup()
    from utils.google_sheets import test_connection, create_or_find_sheet, sync_questions_to_sheets
    using_django = True
except:
    using_django = False

def test_env_variables():
    """Test if the required environment variables are set"""
    sheets_enabled = os.environ.get('GOOGLE_SHEETS_ENABLED')
    sheets_id = os.environ.get('GOOGLE_SHEETS_ID')
    
    print("Environment Variables:")
    print(f"GOOGLE_SHEETS_ENABLED: {sheets_enabled}")
    print(f"GOOGLE_SHEETS_ID: {sheets_id}")
    
    if not sheets_enabled or sheets_enabled.lower() != 'true':
        print("WARNING: Google Sheets integration is disabled (GOOGLE_SHEETS_ENABLED != True)")
        return False
        
    if not sheets_id:
        print("ERROR: Google Sheets ID is not set (GOOGLE_SHEETS_ID)")
        return False
    
    return True

def test_credentials():
    """Test if the credentials file exists and is valid"""
    # Look for credentials.json in several possible locations
    possible_paths = [
        os.path.join(BASE_DIR, 'credentials.json'),
        os.path.join(BASE_DIR, 'utils', 'credentials.json'),
        os.path.join(BASE_DIR, '..', 'credentials.json')
    ]
    
    creds_path = None
    for path in possible_paths:
        if os.path.exists(path):
            creds_path = path
            break
    
    if not creds_path:
        print("ERROR: credentials.json file not found")
        print(f"Searched in: {possible_paths}")
        return False
    
    print(f"Found credentials at: {creds_path}")
    
    # Verify the file is valid JSON
    try:
        with open(creds_path, 'r') as f:
            json.load(f)
        print("Credentials file is valid JSON")
        return True
    except json.JSONDecodeError:
        print("ERROR: credentials.json is not valid JSON")
        return False
    except Exception as e:
        print(f"ERROR reading credentials file: {str(e)}")
        return False

def test_with_google_api():
    """Test direct connection to Google Sheets API"""
    try:
        from google.oauth2.service_account import Credentials
        from googleapiclient.discovery import build
        
        # Look for credentials.json in several possible locations
        possible_paths = [
            os.path.join(BASE_DIR, 'credentials.json'),
            os.path.join(BASE_DIR, 'utils', 'credentials.json'),
            os.path.join(BASE_DIR, '..', 'credentials.json')
        ]
        
        creds_path = None
        for path in possible_paths:
            if os.path.exists(path):
                creds_path = path
                break
        
        if not creds_path:
            print("ERROR: credentials.json file not found")
            return False
        
        print(f"Using credentials from: {creds_path}")
        
        # Load credentials
        creds = Credentials.from_service_account_file(
            creds_path,
            scopes=['https://www.googleapis.com/auth/spreadsheets']
        )
        
        # Build API service
        service = build('sheets', 'v4', credentials=creds)
        sheets = service.spreadsheets()
        
        # Get spreadsheet ID from environment
        sheets_id = os.environ.get('GOOGLE_SHEETS_ID')
        
        # Try to get spreadsheet info
        response = sheets.get(spreadsheetId=sheets_id).execute()
        
        print("\nSpreadsheet Information:")
        print(f"Title: {response.get('properties', {}).get('title')}")
        print("Sheets:")
        for sheet in response.get('sheets', []):
            print(f"  - {sheet.get('properties', {}).get('title')}")
        
        # Write a test cell
        import datetime
        test_data = [["Test connection", str(datetime.datetime.now())]]
        
        result = sheets.values().update(
            spreadsheetId=sheets_id,
            range="Sheet1!A1:B1",
            valueInputOption="RAW",
            body={"values": test_data}
        ).execute()
        
        print("\nTest write successful!")
        print(f"Updated {result.get('updatedCells')} cells")
        
        return True
        
    except Exception as e:
        print(f"\nERROR connecting to Google Sheets API: {str(e)}")
        return False

def create_test_sheet():
    """Create a test sheet for questions and answers"""
    if not using_django:
        print("Django not set up, skipping create_test_sheet")
        return False
    
    print("\nCreating/finding required sheets...")
    
    # Create Questions sheet
    questions_result = create_or_find_sheet("Questions")
    print(f"Questions sheet: {questions_result.get('message', 'Error')}")
    
    # Create Answers sheet
    answers_result = create_or_find_sheet("Answers")
    print(f"Answers sheet: {answers_result.get('message', 'Error')}")
    
    return questions_result.get('success', False) and answers_result.get('success', False)

def run_all_tests():
    """Run all connection tests"""
    print("=" * 50)
    print("GOOGLE SHEETS CONNECTION TESTER")
    print("=" * 50)
    
    # Check environment variables
    print("\n[1/4] Checking environment variables...")
    env_ok = test_env_variables()
    
    # Check credentials file
    print("\n[2/4] Checking credentials file...")
    creds_ok = test_credentials()
    
    # Test Google API connection
    print("\n[3/4] Testing Google Sheets API connection...")
    api_ok = test_with_google_api()

    print("\n[4/4] Setting up required sheets...")
    sheets_ok = True
    if using_django and env_ok and creds_ok and api_ok:
        sheets_ok = create_test_sheet()
    elif not using_django:
        print("Django not set up, skipping sheet creation")
    
    # Print summary
    print("\n" + "=" * 50)
    print("TEST RESULTS SUMMARY")
    print("=" * 50)
    print(f"Environment Variables: {'✓ OK' if env_ok else '✗ FAILED'}")
    print(f"Credentials File: {'✓ OK' if creds_ok else '✗ FAILED'}")
    print(f"Google Sheets API: {'✓ OK' if api_ok else '✗ FAILED'}")
    if using_django:
        print(f"Sheet Setup: {'✓ OK' if sheets_ok else '✗ FAILED'}")
    
    if env_ok and creds_ok and api_ok and sheets_ok:
        print("\n✅ ALL TESTS PASSED - Google Sheets integration is working correctly!")
        
        if using_django:

            print("\nSyncing questions to Google Sheets...")
            sync_result = sync_questions_to_sheets()

            else:

    else:
        print("\n❌ SOME TESTS FAILED - Please check the errors above")
