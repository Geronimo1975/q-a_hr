### 2. Document Processing for Fine-tuning

Add models to support document management and fine-tuning:

```python
# In models.py
class TrainingDocument(models.Model):
    DOCUMENT_TYPES = (
        ('policy', 'Policy Document'),
        ('faq', 'FAQ Document'),
        ('guide', 'Training Guide'),
        ('script', 'Conversation Script'),
    )
    
    title = models.CharField(max_length=255)
    description = models.TextField(blank=True)
    document_type = models.CharField(max_length=20, choices=DOCUMENT_TYPES)
    file = models.FileField(upload_to='training_documents/')
    uploaded_at = models.DateTimeField(auto_now_add=True)
    is_processed_for_training = models.BooleanField(default=False)
    
    def __str__(self):
        return self.title
        
class TrainingExample(models.Model):
    """Individual training examples extracted from documents"""
    source_document = models.ForeignKey(TrainingDocument, on_delete=models.CASCADE, related_name='training_examples')
    question = models.TextField()
    answer = models.TextField()
    created_at = models.DateTimeField(auto_now_add=True)
    
    def __str__(self):
        return f"Example from {self.source_document.title}"
```

### 3. Document Processing Backend

Create a background task to process uploaded documents:

```python
# In tasks.py (using Celery or similar)
from celery import shared_task
from .models import TrainingDocument, TrainingExample
from .document_processor import DocumentProcessor

@shared_task
def process_document_for_training(document_id):
    """Process uploaded document and extract training examples"""
    try:
        document = TrainingDocument.objects.get(id=document_id)
        processor = DocumentProcessor()
        
        # Extract text from document
        text = processor.extract_text_from_document(document.file.path)
        
        # Process based on document type
        if document.document_type == 'faq':
            # Extract Q&A pairs from FAQ document
            qa_pairs = processor.extract_qa_pairs(text)
            for question, answer in qa_pairs:
                TrainingExample.objects.create(
                    source_document=document,
                    question=question,
                    answer=answer
                )
        elif document.document_type == 'policy':
            # Extract policy statements and create examples
            statements = processor.extract_policy_statements(text)
            for statement in statements:
                # Create examples for common questions about this policy
                TrainingExample.objects.create(
                    source_document=document,
                    question=f"What is your policy on {statement.topic}?",
                    answer=statement.content
                )
        # Add other document type processing as needed
        
        # Mark document as processed
        document.is_processed_for_training = True
        document.save()
        
        # Generate fine-tuning file if needed
        create_fine_tuning_file()
    except Exception as e:
        print(f"Error processing document {document_id}: {e}")
        # Log error or send notification
```

### 5. Fine-tuning Workflow Integration

Add functionality to generate fine-tuning files for OpenAI:

```python
# In fine_tuning.py
import json
import os
from django.conf import settings
from .models import TrainingExample

def create_fine_tuning_file():
    """Generate JSONL file for OpenAI fine-tuning"""
    examples = TrainingExample.objects.all()
    
    fine_tuning_data = []
    for example in examples:
        fine_tuning_data.append({
            "messages": [
                {"role": "system", "content": "You are a helpful assistant for a delivery driver recruitment platform."},
                {"role": "user", "content": example.question},
                {"role": "assistant", "content": example.answer}
            ]
        })
    
    # Save to JSONL file
    fine_tuning_file_path = os.path.join(settings.MEDIA_ROOT, 'fine_tuning_data.jsonl')
    with open(fine_tuning_file_path, 'w') as f:
        for item in fine_tuning_data:
            f.write(json.dumps(item) + '\n')
    
    return fine_tuning_file_path
```

### 6. Integration with OpenAI Fine-tuning API

Add integration with OpenAI's fine-tuning API:

```python
# In fine_tuning.py
import openai
from django.conf import settings

def submit_fine_tuning_job(file_path):
    """Submit fine-tuning job to OpenAI"""
    try:
        openai.api_key = settings.OPENAI_API_KEY
        
        # Upload file
        with open(file_path, 'rb') as f:
            response = openai.File.create(
                file=f,
                purpose='fine-tune'
            )
        file_id = response.id
        
        # Create fine-tuning job
        response = openai.FineTuningJob.create(
            training_file=file_id,
            model="gpt-3.5-turbo",  # or your preferred model
        )
        
        # Store job ID for tracking
        job_id = response.id
        
        return job_id
    except Exception as e:
        print(f"Error submitting fine-tuning job: {e}")
        return None
```

### 7. Admin Fine-tuning Control Panel

Add a custom admin view for managing fine-tuning jobs:

```python
# In admin.py
from django.urls import path
from django.contrib import admin
from django.shortcuts import render, redirect
from django.contrib import messages
from .fine_tuning import create_fine_tuning_file, submit_fine_tuning_job

class ChatbotAdminSite(admin.AdminSite):
    site_header = 'Delivery Driver Recruitment Admin'
    
    def get_urls(self):
        urls = super().get_urls()
        custom_urls = [
            path('fine-tuning/', self.admin_view(self.fine_tuning_view), name='fine-tuning'),
        ]
        return custom_urls + urls
    
    def fine_tuning_view(self, request):
        if request.method == 'POST':
            action = request.POST.get('action')
            
            if action == 'generate_file':
                file_path = create_fine_tuning_file()
                messages.success(request, f"Fine-tuning file generated: {file_path}")
                
            elif action == 'submit_job':
                file_path = create_fine_tuning_file()
                job_id = submit_fine_tuning_job(file_path)
                
                if job_id:
                    messages.success(request, f"Fine-tuning job submitted. Job ID: {job_id}")
                else:
                    messages.error(request, "Failed to submit fine-tuning job")
        
        # Get training data statistics
        from .models import TrainingExample, TrainingDocument
        context = {
            'total_documents': TrainingDocument.objects.count(),
            'processed_documents': TrainingDocument.objects.filter(is_processed_for_training=True).count(),
            'total_examples': TrainingExample.objects.count(),
            'example_distribution': TrainingExample.objects.values('source_document__document_type').annotate(count=Count('id')),
            'title': 'Fine-tuning Management',
        }
        
        return render(request, 'admin/fine_tuning.html', context)

# Replace default admin site
admin_site = ChatbotAdminSite(name='chatbot_admin')
```
